cmake_minimum_required(VERSION 3.22.1)
project(khodroyar_tokenizer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_NFKC_COMPILE OFF CACHE BOOL "" FORCE)
set(SENTENCEPIECE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.cache/sentencepiece")
if(NOT EXISTS "${SENTENCEPIECE_ROOT}/KHODROYAR-SOURCE-PIN")
    message(FATAL_ERROR "Run python tools/offline-albert/prepare_sentencepiece.py before building")
endif()
file(READ "${SENTENCEPIECE_ROOT}/KHODROYAR-SOURCE-PIN" SPM_SOURCE_PIN)
if(NOT SPM_SOURCE_PIN STREQUAL "31646a467d2051eb904e0b45de3a73e91fe1c1e3")
    message(FATAL_ERROR "Unexpected tokenizer source revision")
endif()
add_subdirectory("${SENTENCEPIECE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/sentencepiece" EXCLUDE_FROM_ALL)
add_library(khodroyar_tokenizer SHARED sentencepiece_bridge.cpp)
target_include_directories(khodroyar_tokenizer PRIVATE "${SENTENCEPIECE_ROOT}/src")
target_link_libraries(khodroyar_tokenizer PRIVATE sentencepiece-static log)
target_link_options(khodroyar_tokenizer PRIVATE "-Wl,-z,max-page-size=16384")
